{{ range (split $header ",") }}{{ end }} {{- range seq 1 50 -}} {{- $rowKey := printf "row%d" . -}} {{- $row := .Get $rowKey -}} {{- if $row -}} {{ range (split $row ",") }}{{ end }} {{- end -}} {{- end -}}
{{ . | trim " " }}
{{ . | trim " " }}
{{- else -}}
{{ .Inner | markdownify }}
{{- end -}}
